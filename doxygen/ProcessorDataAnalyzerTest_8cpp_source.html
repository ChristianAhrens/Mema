19// Coverage for Mema::ProcessorDataAnalyzer -- the class Mema.Mo uses to turn the
20// raw audio buffers streamed from Mema into level and spectrum data for its
21// meterbridge/2D-field/waveform/spectrum visualisations. Level computation
22// happens synchronously inside analyzeData() (verified below with an exact,
23// hand-computed expected value), so no audio device or running message loop is
24// needed. Only the level path is verified with exact numbers; the spectrum (FFT)
25// path is exercised as a smoke test (it runs without crashing/NaN-ing and
26// produces normalised values) rather than asserting exact per-bin numbers, since
27// hand-deriving the expected output of the log-frequency-banded FFT would be
28// fragile and error-prone to get right independently of the implementation.
29//
30// NOTE [bug]: both the level and spectrum computations in analyzeData() read
31// their input from m_centiSecondBuffer -- but that buffer is only ever written
32// to (via copyFrom()) inside the `if (isBufferProcessingUsed())` branch. If
33// buffer processing is NOT enabled, level/spectrum silently compute from
34// whatever m_centiSecondBuffer happened to contain before (stale/uninitialised
35// memory, not the actual signal just passed in) rather than from the real
36// input. This is not just a theoretical corner case: MemaProcessor.cpp calls
37// `setUseProcessingTypes(true, false, false)` for its own input/output
38// analyzers (level only, no buffer) -- so Mema's own on-device level metering
39// currently reads garbage, not real levels. MemaMo never hits this because it
40// always pairs level/spectrum with buffer processing (see MemaMoComponent.cpp).
41// The tests below deliberately use MemaMo's real (working) call pattern --
42// buffer processing enabled alongside level/spectrum -- both to get
43// deterministic, correct results and to avoid asserting on undefined behaviour;
44// see the accompanying report for the suggested fix to analyzeData() itself.
45//
46// NOTE: constructing ProcessorDataAnalyzer starts a juce::Timer (setHoldTime() in
47// its constructor), which lazily spins up JUCE's shared background TimerThread.
48// This is standard, supported JUCE usage in a plain console app (Timer::~Timer()
49// stops the timer automatically), but it is the first place in this test suite
50// a Timer-derived object is constructed -- flagging in case it ever behaves
51// differently than juce::JUCEApplication's already-proven-safe dummy-instance
52// pattern used in *AppConfigurationTest.cpp.
53//
54// NOTE: level channels are 1-based (SetLevel(i + 1, ...) in analyzeData(), matching
55// MeterbridgeComponent's `for (i = 1; i <= channelCount; ++i)`), while spectrum
56// channels are 0-based (SetSpectrum(i, ...), matching SpectrumAudioComponent's
57// `for (i = 0; i < numVisibleChannels; ++i)`). Both producer and consumer agree
58// within each data type, so this isn't a functional bug, just an inconsistency
59// between the two conventions worth knowing about if ever correlating a "channel
60// N" between a level display and a spectrum display for the same physical input.

67class ProcessorDataAnalyzerLevelTest : public juce::UnitTest
68{
69public:
70 ProcessorDataAnalyzerLevelTest() : juce::UnitTest ("ProcessorDataAnalyzer level processing", "Mema.Mo") {}
71
72 void runTest() override
73 {
74 beginTest ("analyzeData() is a no-op before initializeParameters()");
75 {
76 ProcessorDataAnalyzer analyzer;
77 expect (! analyzer.IsInitialized());
78
79 juce::AudioBuffer<float> buffer (1, 10);
80 buffer.clear();
81 analyzer.analyzeData (buffer); // should not crash, should not create any channel data
82 expectEquals ((int) analyzer.GetLevel().GetChannelCount(), 0);
83 }
84
85 beginTest ("A constant-amplitude buffer produces the exact expected peak/RMS level");
86 {
87 ProcessorDataAnalyzer analyzer;
88 analyzer.initializeParameters (48000.0, 480);
89 analyzer.setUseProcessingTypes (true, true, false); // level + buffer, matching MemaMoComponent's real usage
90 expect (analyzer.IsInitialized());
91
92 // exactly one centisecond's worth of samples at 48kHz (48000 * 0.01)
93 juce::AudioBuffer<float> buffer (1, 480);
94 for (int s = 0; s < 480; ++s)
95 buffer.setSample (0, s, 0.5f);
96
97 analyzer.analyzeData (buffer);
98
99 // level channels are 1-based -- see file header note
100 auto level = analyzer.GetLevel().GetLevel (1);
101 expectWithinAbsoluteError (level.peak, 0.5f, 1.0e-5f);
102 expectWithinAbsoluteError (level.rms, 0.5f, 1.0e-5f); // RMS of a constant signal equals its magnitude
103 expectWithinAbsoluteError (level.hold, 0.5f, 1.0e-5f); // hold = max(peak, previous hold=0)
104 }
105
106 beginTest ("Hold value stays at the previous peak when a quieter buffer follows (until FlushHold() fires)");
107 {
108 ProcessorDataAnalyzer analyzer;
109 analyzer.initializeParameters (48000.0, 480);
110 analyzer.setUseProcessingTypes (true, true, false); // level + buffer, matching MemaMoComponent's real usage
111
112 juce::AudioBuffer<float> loud (1, 480);
113 for (int s = 0; s < 480; ++s)
114 loud.setSample (0, s, 0.5f);
115 analyzer.analyzeData (loud);
116
117 juce::AudioBuffer<float> silence (1, 480);
118 silence.clear();
119 analyzer.analyzeData (silence);
120
121 auto level = analyzer.GetLevel().GetLevel (1);
122 expectWithinAbsoluteError (level.peak, 0.0f, 1.0e-6f);
123 expectWithinAbsoluteError (level.hold, 0.5f, 1.0e-5f);
124 }
125
126 beginTest ("getGlobalMindB()/getGlobalMaxdB() match the documented -80..0 dB display range");
127 {
128 expectEquals (ProcessorDataAnalyzer::getGlobalMindB(), -80);
129 expectEquals (ProcessorDataAnalyzer::getGlobalMaxdB(), 0);
130 }
131 }
132};

137class ProcessorDataAnalyzerSpectrumSmokeTest : public juce::UnitTest
138{
139public:
140 ProcessorDataAnalyzerSpectrumSmokeTest() : juce::UnitTest ("ProcessorDataAnalyzer spectrum processing (smoke test)", "Mema.Mo") {}
141
142 void runTest() override
143 {
144 beginTest ("Feeding enough samples to trigger at least one FFT produces finite, normalised spectrum values");
145
146 ProcessorDataAnalyzer analyzer;
147 analyzer.initializeParameters (48000.0, 480);
148 analyzer.setUseProcessingTypes (true, true, true); // level + buffer + spectrum, matching MemaMoComponent's real usage
149
150 // fftOrder is 12 (fftSize 4096) internally; feed more than that many silent
151 // samples in one go so analyzeData()'s internal centisecond loop pushes at
152 // least one full FFT window through processSpectrumForChannel().
153 juce::AudioBuffer<float> buffer (1, 4096 + 480);
154 buffer.clear();
155
156 analyzer.analyzeData (buffer);
157
158 // spectrum channels are 0-based -- see file header note
159 const auto& spectrum = analyzer.GetSpectrum().GetSpectrum (0);
160
161 expectEquals (spectrum.minFreq, 20.0f);
162 expect (spectrum.maxFreq > spectrum.minFreq);
163 expect (spectrum.freqRes > 0.0f);
164
165 for (int i = 0; i < ProcessorSpectrumData::SpectrumBands::count; ++i)
166 {
167 expect (! std::isnan (spectrum.bandsPeak[i]) && ! std::isinf (spectrum.bandsPeak[i]), "spectrum band values must be finite");
168 expect (spectrum.bandsPeak[i] >= 0.0f && spectrum.bandsPeak[i] <= 1.0f, "spectrum band values should be normalised to [0, 1]");
169 }
170 }
171};

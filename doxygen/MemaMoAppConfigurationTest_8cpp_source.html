19// Coverage for MemaMoAppConfiguration::isValid() (the static, XmlElement-taking
20// overload) -- the gate that decides whether a loaded/received config file is
21// complete enough to apply. As with Mema's MemaAppConfigurationTest.cpp, this
22// deliberately does NOT construct a full MemaMoAppConfiguration instance (real
23// file I/O + a background flush thread in AppConfigurationBase::InitializeBase()).
24// The static isValid() overload only needs a live juce::JUCEApplication instance,
25// for the application-name tag check in JUCEAppBasics::AppConfigurationBase::isValid().

32 // See MemaAppConfigurationTest.cpp (Mema's Tests project) for why this is safe:
33 // JUCEApplicationBase's constructor just registers the global instance pointer,
34 // the message loop is never started, initialise()/shutdown() are never called.
35 class DummyMemaMoApplication : public juce::JUCEApplication
36 {
37 public:
38 const juce::String getApplicationName() override { return "Mema.Mo"; }
39 const juce::String getApplicationVersion() override { return "0.0.0"; }
40 void initialise (const juce::String&) override {}
41 void shutdown() override {}
42 };

44 // Mirrors the structure MemaMoDefault.config and MemaMoComponent's config dump use.
45 std::unique_ptr<juce::XmlElement> makeValidConfig()
46 {
47 auto root = std::make_unique<juce::XmlElement> ("Mema.Mo");
48
49 auto* connectionConfig = root->createNewChildElement ("CONNECTIONCONFIG");
50 connectionConfig->createNewChildElement ("SERVICEDESCRIPTION");
51
52 auto* visuConfig = root->createNewChildElement ("VISUCONFIG");
53 visuConfig->createNewChildElement ("OUTPUTVISUTYPE");
54 visuConfig->createNewChildElement ("METERINGCOLOUR");
55 visuConfig->createNewChildElement ("LOOKANDFEEL");
56
57 return root;
58 }
59}

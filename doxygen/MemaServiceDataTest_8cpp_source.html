19// Coverage for Mema::ServiceData -- the compile-time constants identifying the
20// Mema multicast service discovery protocol (used by Mema.Mo to find Mema
21// instances on the network). Pure static string/int constants, no dependencies.
22//
23// NOTE: ServiceData::getServiceDescription() is declared in MemaServiceData.h but
24// never defined anywhere in the codebase, and nothing calls it (every real call
25// site uses the unrelated JUCEAppBasics::ServiceTopologyManager::getServiceDescription()
26// instead). It is intentionally not exercised here -- referencing it would fail to
27// link. Worth removing as dead API in a cleanup pass.

34class MemaServiceDataTest : public juce::UnitTest
35{
36public:
37 MemaServiceDataTest() : juce::UnitTest ("MemaServiceData", "Mema.Mo") {}
38
39 void runTest() override
40 {
41 beginTest ("Service type UIDs are built from the common base and are distinct per role");
42
43 auto base = ServiceData::getServiceTypeUIDBase();
44 auto master = ServiceData::getMasterServiceTypeUID();
45 auto monitor = ServiceData::getMonitorServiceTypeUID();
46 auto remote = ServiceData::getRemoteServiceTypeUID();
47
48 expect (master.startsWith (base));
49 expect (monitor.startsWith (base));
50 expect (remote.startsWith (base));
51
52 expect (master != monitor, "master and monitor service UIDs must be distinct");
53 expect (master != remote, "master and remote service UIDs must be distinct");
54 expect (monitor != remote, "monitor and remote service UIDs must be distinct");
55
56 beginTest ("Broadcast and connection ports are distinct, valid, non-privileged ports");
57
58 auto broadcastPort = ServiceData::getBroadcastPort();
59 auto connectionPort = ServiceData::getConnectionPort();
60
61 expect (broadcastPort != connectionPort, "broadcast and connection must use different ports");
62 expect (broadcastPort > 1024 && broadcastPort < 65536, "broadcast port should be a valid non-privileged port");
63 expect (connectionPort > 1024 && connectionPort < 65536, "connection port should be a valid non-privileged port");
64 }
65};

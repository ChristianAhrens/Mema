19// Coverage for Mema::InputPositionMapper -- the core panning algorithm behind
20// Mema.Re's 2D-field control: given an input's (x, y) position and a sharpness
21// setting, it computes a per-output gain for every configured output position.
22// This is pure math (no juce::Component involved), driven entirely through two
23// std::function hooks (getAngleForChannelType, onInputPositionMapped), so it's
24// directly unit-testable.
25//
26// The two synthetic output positions used below (Left at -90 degrees, Right at
27// +90 degrees, i.e. (-1,0) and (1,0)) are chosen to make the expected proximity/
28// level values exact and hand-verifiable, matching the documented algorithm in
29// InputPositionMapper.h -- not because they're how any real Mema speaker layout
30// is actually angled.
31
32#include <JuceHeader.h>
33#include <MemaClientCommon/InputPositionMapper.h>
34
35using namespace Mema;
36
37namespace
38{
39 float angleForTestChannel (const juce::AudioChannelSet::ChannelType& channelType)
40 {
41 // Left at -90 degrees -> (sin(-90), -cos(-90)) = (-1, 0)
42 // Right at +90 degrees -> (sin(90), -cos(90)) = (1, 0)
43 if (channelType == juce::AudioChannelSet::left)
44 return -90.0f;
45 if (channelType == juce::AudioChannelSet::right)
46 return 90.0f;
47 jassertfalse;
48 return 0.0f;
49 }
50}

52class InputPositionMapperTest : public juce::UnitTest
53{
54public:
55 InputPositionMapperTest() : juce::UnitTest ("InputPositionMapper", "Mema.Re") {}
56
57 void runTest() override
58 {
59 beginTest ("Without both hooks set, mapInputPosition() is a safe no-op");
60 {
61 InputPositionMapper mapper;
62 mapper.setOutputIncludePositions ({ juce::AudioChannelSet::left });
63 // neither hook set -- should not crash, callback simply never fires
64 mapper.mapInputPosition (1, { 0.0f, 0.0f }, 0.0f);
65 }
66
67 beginTest ("Input exactly at a single output's position yields full level for that output");
68 {
69 InputPositionMapper mapper;
70 mapper.getAngleForChannelType = angleForTestChannel;
71 mapper.setOutputIncludePositions ({ juce::AudioChannelSet::left });
72
73 std::map<juce::AudioChannelSet::ChannelType, float> result;
74 mapper.onInputPositionMapped = [&] (std::uint16_t, const std::map<juce::AudioChannelSet::ChannelType, float>& dists)
75 {
76 result = dists;
77 };
78
79 mapper.mapInputPosition (1, { -1.0f, 0.0f }, 0.0f); // exactly on Left's position, sharpness 0
80
81 expectWithinAbsoluteError (result[juce::AudioChannelSet::left], 1.0f, 1.0e-5f);
82 }
83
84 beginTest ("Centred input with zero sharpness splits level equally between two symmetric outputs");
85 {
86 InputPositionMapper mapper;
87 mapper.getAngleForChannelType = angleForTestChannel;
88 mapper.setOutputIncludePositions ({ juce::AudioChannelSet::left, juce::AudioChannelSet::right });
89
90 std::map<juce::AudioChannelSet::ChannelType, float> result;
91 mapper.onInputPositionMapped = [&] (std::uint16_t, const std::map<juce::AudioChannelSet::ChannelType, float>& dists)
92 {
93 result = dists;
94 };
95
96 mapper.mapInputPosition (1, { 0.0f, 0.0f }, 0.0f); // field centre, no sharpness
97
98 // distance from centre to either output is 1.0 on the unit circle -> proximity 0.5,
99 // and with sharpness 0 the exponent is 1, so the level is exactly 0.5 for both.
100 expectWithinAbsoluteError (result[juce::AudioChannelSet::left], 0.5f, 1.0e-5f);
101 expectWithinAbsoluteError (result[juce::AudioChannelSet::right], 0.5f, 1.0e-5f);
102 }
103
104 beginTest ("Maximum sharpness at the field edge gives exact single-output isolation");
105 {
106 InputPositionMapper mapper;
107 mapper.getAngleForChannelType = angleForTestChannel;
108 mapper.setOutputIncludePositions ({ juce::AudioChannelSet::left, juce::AudioChannelSet::right });
109
110 std::map<juce::AudioChannelSet::ChannelType, float> result;
111 mapper.onInputPositionMapped = [&] (std::uint16_t, const std::map<juce::AudioChannelSet::ChannelType, float>& dists)
112 {
113 result = dists;
114 };
115
116 // exactly at Left's position (radius 1.0) with sharpness 1.0 -> hardPanBlend reaches
117 // exactly 1.0, so the result must be the *exact* hard values, not just close to them.
118 mapper.mapInputPosition (1, { -1.0f, 0.0f }, 1.0f);
119
120 expectEquals (result[juce::AudioChannelSet::left], 1.0f);
121 expectEquals (result[juce::AudioChannelSet::right], 0.0f);
122 }
123
124 beginTest ("Maximum sharpness at the field centre does NOT force single-output isolation");
125 {
126 // Documented behaviour: hardPanBlend is gated by radius, so even sharpness == 1.0
127 // still blends across outputs when the input sits at the centre (radius 0).
128 InputPositionMapper mapper;
129 mapper.getAngleForChannelType = angleForTestChannel;
130 mapper.setOutputIncludePositions ({ juce::AudioChannelSet::left, juce::AudioChannelSet::right });
131
132 std::map<juce::AudioChannelSet::ChannelType, float> result;
133 mapper.onInputPositionMapped = [&] (std::uint16_t, const std::map<juce::AudioChannelSet::ChannelType, float>& dists)
134 {
135 result = dists;
136 };
137
138 mapper.mapInputPosition (1, { 0.0f, 0.0f }, 1.0f); // centre, max sharpness
139
140 // proximity 0.5 for both, exponent jmap(1, 1, 5) = 5 -> 0.5^5 = 0.03125, and
141 // hardPanBlend == 0 here (radius 0), so the smooth value must survive untouched.
142 expectWithinAbsoluteError (result[juce::AudioChannelSet::left], 0.03125f, 1.0e-5f);
143 expectWithinAbsoluteError (result[juce::AudioChannelSet::right], 0.03125f, 1.0e-5f);
144 expect (result[juce::AudioChannelSet::left] < 1.0f, "centre position must never reach exact single-output isolation, even at max sharpness");
145 }
146
147 beginTest ("Outputs in the ignore list are always forced to zero");
148 {
149 InputPositionMapper mapper;
150 mapper.getAngleForChannelType = angleForTestChannel;
151 mapper.setOutputIncludePositions ({ juce::AudioChannelSet::left });
152 mapper.setOutputIgnorePositions ({ juce::AudioChannelSet::right });
153
154 std::map<juce::AudioChannelSet::ChannelType, float> result;
155 mapper.onInputPositionMapped = [&] (std::uint16_t, const std::map<juce::AudioChannelSet::ChannelType, float>& dists)
156 {
157 result = dists;
158 };
159
160 mapper.mapInputPosition (1, { -1.0f, 0.0f }, 1.0f);
161
162 expectEquals (result[juce::AudioChannelSet::right], 0.0f);
163 }
164
165 beginTest ("The channel argument is forwarded unchanged to the callback");
166 {
167 InputPositionMapper mapper;
168 mapper.getAngleForChannelType = angleForTestChannel;
169 mapper.setOutputIncludePositions ({ juce::AudioChannelSet::left });
170
171 std::uint16_t gotChannel = 0;
172 mapper.onInputPositionMapped = [&] (std::uint16_t channel, const std::map<juce::AudioChannelSet::ChannelType, float>&)
173 {
174 gotChannel = channel;
175 };
176
177 mapper.mapInputPosition (7, { 0.0f, 0.0f }, 0.0f);
178
179 expectEquals ((int) gotChannel, 7);
180 }
181 }
182};

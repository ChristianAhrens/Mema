29// Converts a normalised Mema value to the native range expected by ParameterControlComponent.
30// Toggle: 0/1 unchanged.
31// Discrete: normalised 0-1 → zero-based step index as float.
32// Continuous: normalised 0-1 → [minValue, maxValue].
33static float normalisedToNative(const Mema::PluginParameterInfo& pi, float normalisedValue)
34{
35 switch (pi.type)
36 {
37 case Mema::ParameterControlType::Discrete:
38 return pi.stepCount > 1 ? normalisedValue * float(pi.stepCount - 1) : 0.0f;
39 case Mema::ParameterControlType::Continuous:
40 return pi.minValue + normalisedValue * (pi.maxValue - pi.minValue);
41 default: // Toggle
42 return normalisedValue;
43 }
44}
45
46// Converts a native ParameterControlComponent value back to the normalised range Mema uses.
47static float nativeToNormalised(const Mema::PluginParameterInfo& pi, float nativeValue)
48{
49 switch (pi.type)
50 {
51 case Mema::ParameterControlType::Discrete:
52 return pi.stepCount > 1 ? nativeValue / float(pi.stepCount - 1) : 0.0f;
53 case Mema::ParameterControlType::Continuous:
54 {
55 const float range = pi.maxValue - pi.minValue;
56 return range > 0.0f ? (nativeValue - pi.minValue) / range : 0.0f;
57 }
58 default: // Toggle
59 return nativeValue;
60 }
61}

63// Builds a ParameterControlInfo (native range) from a PluginParameterInfo (normalised).
64static JUCEAppBasics::ParameterControlInfo toParameterControlInfo(const Mema::PluginParameterInfo& src)
65{
66 JUCEAppBasics::ParameterControlInfo info;
67 info.index = src.index;
68 info.name = src.name;
69 info.type = static_cast<JUCEAppBasics::ParameterControlType>(static_cast<int>(src.type));
70 info.minValue = src.minValue;
71 info.maxValue = src.maxValue;
72 info.stepSize = src.stepSize;
73 info.stepCount = src.stepCount;
74 info.stepNames = src.stepNames;
75 info.currentValue = normalisedToNative(src, src.currentValue);
76 return info;
77}

80PluginControlComponent::PluginControlComponent()
81 : MemaClientControlComponentBase()
82{
83 auto noconfigui = juce::JUCEApplication::getInstance()->getCommandLineParameters().contains("--noconfigui");
84
85 m_enableButton = std::make_unique<juce::DrawableButton>("pluginEnable", juce::DrawableButton::ImageOnButtonBackground);
86 m_enableButton->setClickingTogglesState(true);
87 m_enableButton->setTooltip("Toggle plugin processing on/off");
88 m_enableButton->onStateChange = [=]() {
89 if (onPluginEnabledChanged)
90 onPluginEnabledChanged(m_enableButton->getToggleState());
91 };
92 if (noconfigui)
93 addChildComponent(m_enableButton.get());
94 else
95 addAndMakeVisible(m_enableButton.get());
96
97 m_prePostButton = std::make_unique<juce::TextButton>("Post", "Toggle plugin pre/post matrix insertion");
98 m_prePostButton->setClickingTogglesState(true);
99 m_prePostButton->onStateChange = [=]() {
100 if (onPluginPrePostChanged)
101 onPluginPrePostChanged(m_prePostButton->getToggleState());
102 };
103 if (noconfigui)
104 addChildComponent(m_prePostButton.get());
105 else
106 addAndMakeVisible(m_prePostButton.get());
107
108 m_pluginNameLabel = std::make_unique<juce::Label>("pluginName");
109 m_pluginNameLabel->setFont(m_pluginNameLabel->getFont().withHeight(25));
110 m_pluginNameLabel->setJustificationType(juce::Justification::centredLeft);
111 addAndMakeVisible(m_pluginNameLabel.get());
112
113 m_paramCtrl = std::make_unique<JUCEAppBasics::ParameterControlComponent>();
114 m_paramCtrl->onParameterValueChanged = [this](int idx, float nativeValue) {
115 if (idx < 0 || static_cast<std::size_t>(idx) >= m_pluginParameterInfos.size())
116 return;
117 auto& pi = m_pluginParameterInfos[static_cast<std::size_t>(idx)];
118 if (!pi.isRemoteControllable)
119 return;
120
121 const float normalisedValue = nativeToNormalised(pi, nativeValue);
122 pi.currentValue = normalisedValue;
123
124 if (onPluginParameterValueChanged)
125 onPluginParameterValueChanged(static_cast<std::uint16_t>(idx),
126 pi.id.toStdString(),
127 normalisedValue);
128 };
129 addAndMakeVisible(m_paramCtrl.get());
130}

141void PluginControlComponent::resized()
142{
143 auto margin = 25;
144 auto bounds = getLocalBounds();
145 auto headerRow = bounds.removeFromTop(margin);
146
147 const bool enableVisible = m_enableButton && m_enableButton->isVisible();
148 const bool prePostVisible = m_prePostButton && m_prePostButton->isVisible();
149
150 if (enableVisible || prePostVisible)
151 {
152 const int enableBtnWidth = margin;
153 const int prePostBtnWidth = 2 * margin;
154
155 int totalBtnsWidth = 0;
156 if (enableVisible) totalBtnsWidth += enableBtnWidth;
157 if (prePostVisible) totalBtnsWidth += prePostBtnWidth;
158
159 const int labelWidth = juce::jmax(80, headerRow.getWidth() / 2);
160 const int groupWidth = totalBtnsWidth + labelWidth;
161
162 const int groupX = headerRow.getX() + juce::jmax(0, (headerRow.getWidth() - groupWidth) / 2);
163 auto groupBounds = juce::Rectangle<int>(groupX, headerRow.getY(),
164 juce::jmin(groupWidth, headerRow.getWidth()),
165 headerRow.getHeight());
166
167 if (enableVisible)
168 m_enableButton->setBounds(groupBounds.removeFromLeft(enableBtnWidth).reduced(2));
169 if (prePostVisible)
170 m_prePostButton->setBounds(groupBounds.removeFromLeft(prePostBtnWidth).reduced(2));
171 if (m_pluginNameLabel)
172 m_pluginNameLabel->setBounds(groupBounds);
173 }
174 else
175 {
176 if (m_pluginNameLabel)
177 m_pluginNameLabel->setBounds(headerRow);
178 }
179
180 bounds.removeFromLeft(margin);
181 bounds.removeFromRight(margin);
182 m_parameterBounds = bounds;
183
184 if (m_paramCtrl)
185 m_paramCtrl->setBounds(m_parameterBounds);
186}

243void PluginControlComponent::setParameterInfos(const std::vector<Mema::PluginParameterInfo>& parameterInfos)
244{
245 if (m_pluginParameterInfos != parameterInfos)
246 {
247 m_pluginParameterInfos = parameterInfos;
248
249 if (m_paramCtrl)
250 {
251 std::vector<JUCEAppBasics::ParameterControlInfo> infos;
252 infos.reserve(m_pluginParameterInfos.size());
253 for (const auto& pi : m_pluginParameterInfos)
254 if (pi.isRemoteControllable)
255 infos.push_back(toParameterControlInfo(pi));
256 m_paramCtrl->setParameters(infos);
257 }
258 }
259}

261void PluginControlComponent::setParameterValue(std::uint16_t index, std::string id, float normalisedValue)
262{
263 if (index >= m_pluginParameterInfos.size())
264 return;
265
266 m_pluginParameterInfos[index].currentValue = normalisedValue;
267 m_pluginParameterInfos[index].id = id;
268
269 if (m_paramCtrl)
270 m_paramCtrl->setParameterValue(static_cast<int>(index),
271 normalisedToNative(m_pluginParameterInfos[index], normalisedValue));
272}

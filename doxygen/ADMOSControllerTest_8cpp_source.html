19// Coverage for Mema::ADMOSController -- the ADM-OSC protocol implementation behind
20// Mema.Re's external panning control (see README's "Mema.Re ADM-OSC external
21// control"). Deliberately never calls startConnection(): that's the only method
22// that actually binds a UDP socket / starts a background thread (verified by
23// reading JUCE's OSCReceiver::Pimpl -- its constructor is inert, startThread()
24// only runs from connect()), so every test here only constructs the controller
25// and exercises pure parsing/building/caching logic.
26//
27// setParameter() (called internally by oscMessageReceived()) does post an
28// ADMOSCParameterChangedMessage via juce::MessageListener::postMessage(). The
29// cache write it triggers happens synchronously before that post, so checking
30// getParameter() right after calling oscMessageReceived() is a reliable way to
31// verify address-pattern parsing without depending on a running message loop to
32// ever actually deliver that notification.
33

40class ADMOSCParameterBitPackingTest : public juce::UnitTest
41{
42public:
43 ADMOSCParameterBitPackingTest() : juce::UnitTest ("ADMOSCParameter bit-packing round-trip", "Mema.Re") {}
44
45 void runTest() override
46 {
47 beginTest ("X/Y/Z/Width round-trip a float value exactly");
48 {
49 expectEquals (ADMOSController::ADMOSCParameterX (0.25f).getParameterVal(), 0.25f);
50 expectEquals (ADMOSController::ADMOSCParameterY (-0.5f).getParameterVal(), -0.5f);
51 expectEquals (ADMOSController::ADMOSCParameterZ (1.0f).getParameterVal(), 1.0f);
52 expectEquals (ADMOSController::ADMOSCParameterWidth (0.75f).getParameterVal(), 0.75f);
53 }
54
55 beginTest ("XY/XYZ round-trip both/all components exactly");
56 {
57 auto xy = ADMOSController::ADMOSCParameterXY (0.1f, 0.2f).getParameterVals();
58 expectEquals (std::get<0> (xy), 0.1f);
59 expectEquals (std::get<1> (xy), 0.2f);
60
61 auto xyz = ADMOSController::ADMOSCParameterXYZ (0.1f, 0.2f, 0.3f).getParameterVals();
62 expectEquals (std::get<0> (xyz), 0.1f);
63 expectEquals (std::get<1> (xyz), 0.2f);
64 expectEquals (std::get<2> (xyz), 0.3f);
65 }
66
67 beginTest ("Mute round-trips both bool and 0/1 int forms");
68 {
69 expect (ADMOSController::ADMOSCParameterMute (true).getParameterVal());
70 expect (! ADMOSController::ADMOSCParameterMute (false).getParameterVal());
71 expectEquals (ADMOSController::ADMOSCParameterMute (1).getParameterVal01(), 1);
72 expectEquals (ADMOSController::ADMOSCParameterMute (0).getParameterVal01(), 0);
73 }
74
75 beginTest ("Narrowing conversions extract the matching component from a wider parameter");
76 {
77 // e.g. an XYZ update should still yield the correct Z value when narrowed to
78 // just ADMOSCParameterZ -- this is how setParameter()'s cache-then-narrow usage works.
79 ADMOSController::ADMOSCParameterXYZ xyz (0.4f, 0.5f, 0.6f);
80 ADMOSController::ADMOSCParameterZ z (xyz);
81 expectEquals (z.getParameterVal(), 0.6f);
82 }
83 }
84};

188class ADMOSControllerIncomingMessageTest : public juce::UnitTest
189{
190public:
191 ADMOSControllerIncomingMessageTest() : juce::UnitTest ("ADMOSController incoming OSC message parsing", "Mema.Re") {}
192
193 void runTest() override
194 {
195 beginTest ("An /xy message updates the cached XY parameter for the addressed object");
196 {
197 ADMOSController controller;
198 juce::OSCMessage msg (juce::OSCAddressPattern ("/adm/obj/3/xy"), 0.25f, 0.75f);
199 controller.oscMessageReceived (msg);
200
201 auto cached = ADMOSController::ADMOSCParameterXY (controller.getParameter (3, ADMOSController::ADMOSCParameterType::XY));
202 auto vals = cached.getParameterVals();
203 expectEquals (std::get<0> (vals), 0.25f);
204 expectEquals (std::get<1> (vals), 0.75f);
205 }
206
207 beginTest ("A /mute message updates the cached mute parameter for the addressed object");
208 {
209 ADMOSController controller;
210 juce::OSCMessage msg (juce::OSCAddressPattern ("/adm/obj/2/mute"), (juce::int32) 1);
211 controller.oscMessageReceived (msg);
212
213 auto cached = ADMOSController::ADMOSCParameterMute (controller.getParameter (2, ADMOSController::ADMOSCParameterType::Mute));
214 expect (cached.getParameterVal());
215 }
216
217 beginTest ("A wildcard object identifier applies the update to every known object");
218 {
219 ADMOSController controller;
220 controller.setNumObjects (3);
221
222 juce::OSCMessage msg (juce::OSCAddressPattern ("/adm/obj/*/x"), 0.42f);
223 controller.oscMessageReceived (msg);
224
225 for (int obj = 1; obj <= 3; ++obj)
226 {
227 auto cached = ADMOSController::ADMOSCParameterX (controller.getParameter (obj, ADMOSController::ADMOSCParameterType::X));
228 expectEquals (cached.getParameterVal(), 0.42f);
229 }
230 }
231
232 beginTest ("An unrecognised address pattern is ignored, not crashed on");
233 {
234 ADMOSController controller;
235 juce::OSCMessage msg (juce::OSCAddressPattern ("/completely/unrelated/address"), 1.0f);
236 controller.oscMessageReceived (msg); // should simply do nothing
237 }
238 }
